```typescript
// 保存事件数据到 o_event 和 o_eventChapter 表
for (const event of totalEvent) {
  if (event.event) {
    const chapter = allChapters.find((c) => c.id === event.id);
    
    // PostgreSQL 需要使用 returning 获取插入后的 ID
    const insertResult = await u.db("o_event")
      .insert({
        novelId: event.id,
        name: event.event.slice(0, 100) || "事件",
        content: event.event,
        order: chapter?.order || 0,
      })
      .returning("id");
    const eventId = insertResult[0]?.id;
    
    if (eventId) {
      await u.db("o_eventChapter").insert({
        eventId: eventId,
        novelId: event.id,
        name: event.event.slice(0, 100) || "事件章节",
        content: event.event,
        order: chapter?.order || 0,
      });
    }
  }
}
```

## 三、问题修复建议

### 3.1 `o_event` / `o_eventChapter` 表

**状态**: ✅ **已修复**

**问题**: PostgreSQL 的 `insert` 返回值格式与 SQLite 不同，导致 `eventId` 为 `undefined`

**修复位置**: [`src/utils/cleanNovel.ts`](src/utils/cleanNovel.ts#L88-L116)

**修复方案**: 使用 `.returning("id")` 获取插入后的 ID

```typescript
// PostgreSQL 需要使用 returning 获取插入后的 ID
const insertResult = await u.db("o_event")
  .insert({...})
.returning("id");
const eventId = insertResult[0]?.id;

if (eventId) {
  await u.db("o_eventChapter").insert({...});
}
```

### 3.2 `o_prompt` 表

**状态**: ✅ **已修复**

**问题**: fixDB 中只有 update 没有 insert

**修复位置**: [`src/lib/fixDB.ts`](src/lib/fixDB.ts#L403-L430)

**修复方案**: 先检查后插入

```typescript
for (const prompt of defaultPrompts) {
  const exists = await db("o_prompt").where("type", prompt.type).first();
  if (!exists) {
    await db("o_prompt").insert({
      ...prompt,
      useData: null,
      created_at: new Date(),
      updated_at: new Date(),
    });
  } else {
    await db("o_prompt").where("type", prompt.type).update({
      data: prompt.data,
      name: prompt.name,
      updated_at: new Date(),
    });
  }
}
```

## 四、验证 SQL

```sql
-- 检查 o_event 表数据
SELECT 
  e.id as eventId,
  e.name as eventName,
  e.content,
  ec.id as chapterId,
  n.chapter as novelChapter
FROM o_event e
LEFT JOIN o_eventChapter ec ON ec.eventId = e.id
LEFT JOIN o_novel n ON n.id = ec.novelId;

-- 检查 o_prompt 表数据
SELECT 
  id,
  type,
  name,
  CASE 
    WHEN useData IS NOT NULL THEN 'useData: ' || LENGTH(useData) || ' chars'
    ELSE 'data: ' || LENGTH(data) || ' chars'
  END as contentInfo,
  created_at
FROM o_prompt
ORDER BY type;
```
